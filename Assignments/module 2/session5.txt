(C) Create a Flipkart discount calculator that asks the user for the total cart amount. Use nested if statements to check: if amount > 2000, apply 20% discount; else if amount > 1000, apply 10% discount; else, no discount. Print the final amount to pay.<br><br><em><strong>Hint:</strong> Use nested ifs to check each discount slab.</em>

->

#include <stdio.h>
#include <conio.h>

void main()
{
    float amount = 2500;   // change this value to test
    float discount = 0, finalAmount;

    if (amount > 2000)
    {
        discount = amount * 0.20;
    }
    else
    {
        if (amount > 1000)
        {
            discount = amount * 0.10;
        }
        else
        {
            discount = 0;
        }
    }

    finalAmount = amount - discount;

    printf("Cart Amount: Rs. %.2f\n", amount);
    printf("Discount: Rs. %.2f\n", discount);
    printf("Final Amount to Pay: Rs. %.2f\n", finalAmount);
    
    getch();
}
